if (data) { state.currentUser = { ...data, role: 'visitor' }; document.getElementById('login-view').classList.add('hidden'); document.getElementById('app-view').classList.remove('hidden'); document.getElementById('user-display-name').innerText = data.fullname; const adminKeys = document.getElementById('tbody-admin-keys'); if(adminKeys) { adminKeys.innerHTML = state.keys.map(k => ` ${k.key_name} ${k.location || '-'} ${k.key_uid} ${k.status} `).join(''); } } function renderAdminUsers() { const tbody = document.getElementById('tbody-admin-users');
